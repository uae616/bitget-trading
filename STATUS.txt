================================================================================
KEY CONFIGURATION
================================================================================

Location: config.toml

Bot Settings:
  cycle_seconds = 1              (Check every second)
  max_cycles = 500               (Run 500 cycles, 0=infinite)
  dry_run = false                (false=live trading, true=simulate)
  log_level = "INFO"             (Logging level)

Portfolio:
  coins = ["BTC", "ETH", "SOL", "BNB"]  (Which coins to trade)
  quote_asset = "USDT"                  (Quote currency)

Risk Management:
  daily_loss_limit_pct = 0.03    (Halt if lose 3% in a day)
  min_buy_usdt = 5.1             (Minimum buy amount)
  max_trade_per_symbol_cooldown_sec = 300  (Wait 5min between trades)

Strategy:
  momentum {
    short_min = 30               (30-minute momentum)
    mid_min = 240                (4-hour momentum)
    long_min = 1440              (1-day momentum)
    w_short = 0.5, w_mid = 0.3, w_long = 0.2  (Weights)
  }
